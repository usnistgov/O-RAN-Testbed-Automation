=AVERAGE(D2:D13) & " ± " & T.INV(1-(1-0.95)/2, COUNT(D2:D13)-1) * STDEV.S(D2:D13)/SQRT(COUNT(D2:D13))
=AVERAGE(RANGE) & " ± " & T.INV(1-(1-0.95)/2, COUNT(RANGE)-1) * STDEV.S(RANGE)/SQRT(COUNT(RANGE))
=AVERAGE(RANGE) & " ± " & T.INV(1-(1-0.95)/2, COUNT(RANGE)-1) * STDEV.S(RANGE)/SQRT(COUNT(RANGE))
